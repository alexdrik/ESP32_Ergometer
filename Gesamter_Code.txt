const uint32_t COL_WIDTH_PX = 64;

// Super einfache, universelle Hilfsfunktion zum Beschreiben der Display-Matrix!
void setField(int32_t row, int32_t col, const char *label, const char *valueText)
{
  if (row < MATRIX_ROWS && col < MATRIX_COLS)
    {
      // Baut das Feld direkt im Format "LABEL:WERT" (z.B. "RPM :  85") zusammen
      snprintf(systemData.displayMatrix[row][col],
               COL_CHAR_LIMIT + 1,
               "%-4s:%5s",
               label,
               valueText);
    }
}

void showDisplayDiag()
{
  static char lastMatrix[MATRIX_ROWS][MATRIX_COLS][COL_CHAR_LIMIT + 1] = {0};

  tft.setTextSize(1);
  tft.setTextColor(TFT_GREEN, TFT_BLACK);

  for (int32_t r = 0; r < MATRIX_ROWS; r++)
    {
      for (int32_t c = 0; c < MATRIX_COLS; c++)
        {
          if (strcmp(systemData.displayMatrix[r][c], lastMatrix[r][c]) != 0)
            {
              uint32_t x = c * COL_WIDTH_PX;
              uint32_t y = r * ROW_HEIGHT_PX;

              tft.drawString(systemData.displayMatrix[r][c], x, y);

              strncpy(lastMatrix[r][c], systemData.displayMatrix[r][c], COL_CHAR_LIMIT);
              lastMatrix[r][c][COL_CHAR_LIMIT] = '\0';
            }
        }
    }
}

void showDiagData()

  static uint32_t lastDiagTime = 0;
  if (millis() - lastDiagTime > DIAG_INTERVAL)

// ==============================================================================
// Interrupt-Routine fÃ¼r den Kurbelsensor (Trittfrequenz)
void IRAM_ATTR cadenceISR()
{
  uint32_t now = micros();
  // Entprellung: Ein Trittfrequenz-Signal kann physikalisch nicht schneller als alle 150ms kommen (ca. 400 RPM)
  if (now - gpioData.isrLastCadenceTime > (DEBOUNCE_DELAY * 1000))
    {
      gpioData.isrCadenceDeltaUs = now - gpioData.isrLastCadenceTime;
      gpioData.isrLastCadenceTime = now;
      gpioData.isrPedalDetected = true;
    }
}

// ==============================================================================
// Interrupt-Routine fÃ¼r den Inkrementalgeber (Schritt-ZÃ¤hler Stellmotor)
void IRAM_ATTR encoderISR()
{
  gpioData.isrEncoderCount++;
}

// ==============================================================================
// Interrupt-Routine fÃ¼r den Referenz-Endschalter (Nullpunkt Stellmotor)
void IRAM_ATTR referenceISR()
{
  gpioData.isrEncoderCount = 0; // Sofortige Kalibrierung der Motor-Ist-Position
}

// ==============================================================================
void getGPIOInputs()
{
  uint32_t nowMs = millis();
  uint32_t nowMicros = micros();

  // --- 1. PHYSALISCHE LIVE-ZUSTÃ„NDE DER HARDWARE EINLESEN ---
  // Reed-Sensor, Inkrementalgeber und Endschalter direkt von den Pins abfragen
  gpioData.cadenceRawPulse = (digitalRead(KADENCE_PIN) == LOW);
  gpioData.encoderRawPulse = (digitalRead(INKREMENTAL_PIN) == LOW);
  gpioData.referenceSwitchPressed = (digitalRead(REFERENCE_PIN) == LOW);
  gpioData.plusBtnPressed = (digitalRead(GEAR_PLUS) == LOW);
  gpioData.minusBtnPressed = (digitalRead(GEAR_MINUS) == LOW);

  // --- 2. KADENZ BERECHNEN ---
  if (now - gpioData.isrLastCadenceTime > 3000000)
    {
      systemData.cadenceRPM = 0;
    }
  else if (gpioData.isrCadenceDeltaUs > 0)
    {
      systemData.cadenceRPM = 60000000 / isrCadenceDeltaUs;
    }

  // Pedal-Impuls-Flanke aus der ISR an den Kern Ã¼bergeben
  gpioData.pedalPulseDetected = isrPedalDetected;
  gpioData.isrPedalDetected = false;

  // --- 3. GANGSCHALTUNG (Prellfrei) ---
  if (nowMs - gpioData.lastGearChangeTime > DEBOUNCE_DELAY)
    {
      if (gpioData.plusBtnPressed)
        {
          if (systemData.currentGear < GEAR_COUNT)
            {
              systemData.currentGear++;
              Serial.printf("-> Hochschalten! Neuer Gang: %d\n", systemData.currentGear);
              gpioData.lastGearChangeTime = nowMs;
            }
        }
      else if (gpioData.minusBtnPressed)
        {
          if (systemData.currentGear > 1)
            {
              systemData.currentGear--;
              Serial.printf("-> Runterschalten! Neuer Gang: %d\n", systemData.currentGear);
              gpioData.lastGearChangeTime = nowMs;
            }
        }
    }

  // --- 4. IST-MOTORPOSITION SPEICHERN ---
  gpioData.actualMotorPosition = (int32_t)isrEncoderCount;
}

static BLEAdvertisedDevice *myDevice;
static BLEClient *pClient = nullptr;
static bool doConnect = false;

static void hrNotifyCallback(
    BLERemoteCharacteristic *pBLERemoteCharacteristic,
    uint8_t *pData,
    size_t length,
    bool isNotify)
{
  if (length >= 2)
    {
      uint8_t flags = pData[0];
      int32_t rawHeartRate = 0;

      if ((flags & 0x01) == 0)
        {
          rawHeartRate = pData[1];
        }
      else
        {
          rawHeartRate = pData[1] | (pData[2] << 8);
        }

      // --- DER REKURSIVE FILTER ---
      if (rawHeartRate > 0)
        {
          // 1. Letzten (Ã¤ltesten) Wert an der Zeigerposition abziehen
          pulsGurtData.hrSum -= pulsGurtData.hrBuffer[pulsGurtData.bufferIndex];

          // 2. Neuen Wert in den Puffer schreiben
          pulsGurtData.hrBuffer[pulsGurtData.bufferIndex] = rawHeartRate;

          // 3. Neuen Wert auf die Summe aufaddieren
          pulsGurtData.hrSum += rawHeartRate;

          // 4. Index fÃ¼r den nÃ¤chsten Wert verschieben (mit RingÃ¼berlauf)
          pulsGurtData.bufferIndex++;
          if (pulsGurtData.bufferIndex >= HR_BUFFER_SIZE)
            {
              pulsGurtData.bufferIndex = 0;
            }

          // 5. Mittelwert berechnen und an Core 1 Ã¼bergeben
          pulsGurtData.heartRateBPM = rawHeartRate;
          pulsGurtData.heartRateAverage = pulsGurtData.hrSum / HR_BUFFER_SIZE;
        }
    }
}

class MyAdvertisedDeviceCallbacks : public BLEAdvertisedDeviceCallbacks

  }

// Logik fÃ¼r den Verbindungsaufbau zum Pulsgurt
bool connectToHeartRateSensor()
{
  Serial.println("-> Verbinde mit Pulsgurt...");
  pClient = BLEDevice::createClient();

  if (!pClient->connect(myDevice))
    {
      return false;
    }

  // 1. Herzfrequenz-Dienst abonnieren
  BLERemoteService *pRemoteService = pClient->getService(hrServiceUUID);
  if (pRemoteService != nullptr)
    {
      BLERemoteCharacteristic *pRemoteCharacteristic = pRemoteService->getRemoteCharacteristic(hrCharUUID);
      if (pRemoteCharacteristic != nullptr && pRemoteCharacteristic->canNotify())
        {
          pRemoteCharacteristic->registerForNotify(hrNotifyCallback);
          rxHrConnected = true;
          Serial.println("-> Puls-Kanal erfolgreich abonniert.");
        }
    }

  // 2. Batterieladezustand auslesen
  BLERemoteService *pBatService = pClient->getService(batServiceUUID);
  if (pBatService != nullptr)
    {
      BLERemoteCharacteristic *pBatCharacteristic = pBatService->getRemoteCharacteristic(batCharUUID);
      if (pBatCharacteristic != nullptr && pBatCharacteristic->canRead())
        {
          std::string val = pBatCharacteristic->readValue();
          if (val.length() > 0)
            {
              rxHrBattery = (int32_t)val[0];
              Serial.printf("-> Pulsgurt-Batterie: %d %%\n", rxHrBattery);
            }
        }
    }

  return rxHrConnected;
}

// Diese Funktion rufen wir in deinem WLAN/Bluetooth-Task auf Core 0 auf!
void handleHeartRateConnection()
{
  if (doConnect)
    {
      if (connectToHeartRateSensor())
        {
          Serial.println("-> Pulsgurt erfolgreich gekoppelt!");
        }
      else
        {
          Serial.println("-> Verbindungsaufbau fehlgeschlagen. Starte neuen Suchlauf...");
          BLEDevice::getScan()->start(5, false);
        }
      doConnect = false;
    }

  // Falls die Verbindung abbricht, automatisch den Status zurÃ¼cksetzen und neu suchen
  if (pClient != nullptr && !pClient->isConnected() && rxHrConnected)
    {
      rxHrConnected = false;
      rxHeartRate = 0;
      Serial.println("-> Verbindung zum Pulsgurt verloren. Suche erneut...");
      BLEDevice::getScan()->start(5, false);
    }
}

class ControlPointCallbacks : public BLECharacteristicCallbacks
{
  void onWrite(BLECharacteristic *pCharacteristic)
  {
    std::string value = pCharacteristic->getValue();
    if (value.length() > 0)
      {
        uint8_t opcode = value[0];

        // Opcode 0x00: Request Control (Kinomap fordert die Kontrolle an)
        if (opcode == 0x00)
          {
            uint8_t response[] = {0x80, 0x00, 0x01};
            pControlPointCharacteristic->setValue(response, 3);
            pControlPointCharacteristic->indicate();
          }

        // Opcode 0x05: Set Target Power (ERG-Modus / Wattvorgabe)
        else if (opcode == 0x05 && value.length() >= 3)
          {
            rxTargetPower = (uint32_t)(value[1] | (value[2] << 8));
            rxControlMode = 2; // Modus auf Watt-Steuerung setzen

            uint8_t response[] = {0x80, 0x05, 0x01};
            pControlPointCharacteristic->setValue(response, 3);
            pControlPointCharacteristic->indicate();
          }

void calcCtrlMode()
{
  uint32_t nowMs = millis();

  // --- FALL A: App (Kinomap) ist aktiv verbunden ---
  if (systemData.btClientConnected)
    {
      if (rxControlMode == 1)
        {
          systemData.controlMode = OpMode::SIMU;
        }
      else
        {
          systemData.controlMode = OpMode::MANUAL;
        }
    }
  // --- FALL B: Pulsgurt ist gekoppelt -> Puls-Training ---
  else if (systemData.hrSensorConnected)
    {
      systemData.controlMode = OpMode::PULS;

      // Regelungsentscheidung exakt alle 10 Sekunden ausfÃ¼hren
      if (nowMs - systemData.lastPulsRegulatorTime >= REGULATOR_INTERVAL_MS)
        {
          systemData.lastPulsRegulatorTime = nowMs;

          // Wir nutzen direkt das stabil gefilterte Signal aus dem Input-Modul!
          int32_t stableHR = systemData.heartRateBPM_Average;

          if (stableHR > 0)
            {
              // Puls zu niedrig âž” Gang hochschalten fÃ¼r mehr Last
              if (stableHR <= (CARDIO_TARGET_HR - 10))
                {
                  if (systemData.currentGear < GEAR_COUNT)
                    {
                      systemData.currentGear++;
                      Serial.printf("[CARDIO] Puls stabil niedrig (%d bpm). Schalte HOCH -> Gang %d\n", stableHR, systemData.currentGear);
                    }
                }
              // Puls zu hoch âž” Gang runterschalten fÃ¼r weniger Last
              else if (stableHR >= (CARDIO_TARGET_HR + 10))
                {
                  if (systemData.currentGear > 1)
                    {
                      systemData.currentGear--;
                      Serial.printf("[CARDIO] Puls stabil hoch (%d bpm). Schalte RUNTER -> Gang %d\n", stableHR, systemData.currentGear);
                    }
                }
            }
        }
    }
  // --- FALL C: Manueller Betrieb (Keine Sensoren) ---
  else
    {
      systemData.controlMode = OpMode::MANUAL;
    }
}

// ==============================================================================
// 2. PHYSIKALISCHE GANG- & BREMSMOMENTENBERECHNUNG (calcGear)
// ==============================================================================
void calcGear()
{
  if (systemData.currentGear < 1)
    systemData.currentGear = 1;
  if (systemData.currentGear > GEAR_COUNT)
    systemData.currentGear = GEAR_COUNT;

  float activeRatio = GEAR_RATIOS[systemData.currentGear - 1];
  float wheelCircumferenceMeters = WHEEL_CIRCUMFERENCE_MM / 1000.0f;

  // 1. Virtuelle Geschwindigkeit berechnen
  if (systemData.cadenceRPM > 0)
    {
      systemData.speedKmH = (float)systemData.cadenceRPM * activeRatio * wheelCircumferenceMeters * 0.06f;
    }
  else
    {
      systemData.speedKmH = 0.0f;
    }

  // 2. Millimetergenaue Strecke aufaddieren
  if (systemData.pedalPulseDetected)
    {
      float mmThisRotation = activeRatio * WHEEL_CIRCUMFERENCE_MM;
      systemData.distanceMillimeters += (uint32_t)round(mmThisRotation);
    }

  // 3. Physikalische Leistungsberechnung am Berg & Drehmomente
  if (systemData.cadenceRPM > 0 && systemData.speedKmH > 0.0f)
    {
      float v_ms = systemData.speedKmH / 3.6f;
      float total_power = 0.0f;

      if (systemData.controlMode == OpMode::SIMU)
        {
          float slope_decimal = systemData.targetSlope / 100.0f;

          float p_gravity = TOTAL_SYSTEM_WEIGHT_KG * GRAVITY_G * slope_decimal * v_ms;
          float p_roll = TOTAL_SYSTEM_WEIGHT_KG * GRAVITY_G * CR_ROLL_RESIST * v_ms;
          float p_aero = CW_AERO * (v_ms * v_ms * v_ms);

          total_power = p_gravity + p_roll + p_aero;

          if (total_power < 50.0f)
            total_power = 50.0f;
          if (total_power > 450.0f)
            total_power = 450.0f;
          systemData.targetPowerWatt = (int32_t)round(total_power);
        }
      else
        {
          systemData.targetPowerWatt = 50 + ((systemData.currentGear - 1) * 25);
          total_power = (float)systemData.targetPowerWatt;
        }

      // Drehmoment an der Kurbel (M = P * 60 / (2 * pi * RPM))
      systemData.targetTorqueNm = (total_power * 60.0f) / (2.0f * PI * (float)systemData.cadenceRPM);
    }
  else
    {
      systemData.targetPowerWatt = 0;
      systemData.targetTorqueNm = 0.0f;
    }

  if (systemData.targetTorqueNm > 80.0f)
    {
      systemData.targetTorqueNm = 80.0f;
    }
}

void calcMotorPos()
// TemporÃ¤rer Platzhalter: Lineares Mapping auf deine 24 Motorstufen (MOTOR_ABSPOS)
// Wir nehmen an, dass 0 Nm = Stufe 1 (0 Schritte) und 80 Nm = Stufe 24 (680 Schritte) entspricht.
{

  if (targetLevel < 1)
    targetLevel = 1;
  if (targetLevel > MOTOR_POS_COUNT)
    targetLevel = MOTOR_POS_COUNT;

  systemData.targetMotorPosition = MOTOR_ABSPOS[targetLevel - 1];
}

// Hilfsfunktion: Setzt die PWM-Geschwindigkeit mit einer linearen 100ms-Rampe
void driveMotorPWM(bool forward, bool stopRequest)
{
  uint32_t elapsed = millis() - motorStateStartTime;
  uint32_t duty = 0;

  if (stopRequest)
    {
      if (currentMotorState != MotorState::DECEL && currentMotorState != MotorState::STOPPED)
        {
          currentMotorState = MotorState::DECEL;
          motorStateStartTime = millis();
          elapsed = 0;
        }

      if (currentMotorState == MotorState::DECEL)
        {
          if (elapsed < 100)
            {
              duty = 255 - (elapsed * 255 / 100); // Bremse von 100% auf 0%
            }
          else
            {
              duty = 0;
              currentMotorState = MotorState::STOPPED;
            }
        }
    }
  else
    {
      activeDirectionPos = forward;

      if (currentMotorState == MotorState::STOPPED)
        {
          currentMotorState = MotorState::ACCEL;
          motorStateStartTime = millis();
          elapsed = 0;
        }

      if (currentMotorState == MotorState::ACCEL)
        {
          if (elapsed < 100)
            {
              duty = (elapsed * 255 / 100); // Sanft beschleunigen von 0% auf 100%
            }
          else
            {
              duty = 255;
              currentMotorState = MotorState::RUNNING;
            }
        }
      else if (currentMotorState == MotorState::RUNNING)
        {
          duty = 255; // Volle Kraft
        }
    }

  // PWM-Signale an die DRV8833-KanÃ¤le senden
  if (activeDirectionPos)
    {
      ledcWrite(CH_MOTOR_NEG, 0);
      ledcWrite(CH_MOTOR_POS, duty);
    }
  else
    {
      ledcWrite(CH_MOTOR_POS, 0);
      ledcWrite(CH_MOTOR_NEG, duty);
    }
}

void setGPIOOutputs()
{
  int32_t actualPos = gpioData.actualMotorPosition;
  int32_t finalTarget = gpioData.targetMotorPosition;

  // Absicherung des Zielbereichs
  if (finalTarget < MOTOR_ABSPOS[0])
    finalTarget = MOTOR_ABSPOS[0];
  if (finalTarget > MOTOR_ABSPOS[MOTOR_POS_COUNT - 1])
    finalTarget = MOTOR_ABSPOS[MOTOR_POS_COUNT - 1];

  // --- FALL 1: Referenzfahrt (Homing) ---
  if (systemData.controlMode == OpMode::HOMING)
    {
      if (gpioData.referenceSwitchPressed == false)
        {
          driveMotorPWM(false, false); // Sanft rÃ¼ckwÃ¤rts suchen
        }
      else
        {
          driveMotorPWM(false, true); // Sanft abbremsen
          if (gpioData.currentMotorState == MotorState::STOPPED)
            {
              systemData.controlMode = OpMode::MANUAL;
              systemData.currentGear = 4;
              Serial.println("-> Homing erfolgreich abgeschlossen!");
            }
        }
      return;
    }

  // --- FALL 2: RegulÃ¤rer Betrieb (Zustandsmaschine) ---

  // Zustand A: Getriebespiel-Korrektur lÃ¤uft aktiv im Hintergrund
  if (gpioData.currentMotorState == MotorState::BACKLASH)
    {
      if (actualPos > backlashTargetPos)
        {
          driveMotorPWM(false, false); // Fahre aktiv weiter rÃ¼ckwÃ¤rts
        }
      else
        {
          driveMotorPWM(false, true); // Ziel-Ãœberlauf erreicht -> sanft abbremsen

          if (gpioData.currentMotorState == MotorState::STOPPED)
            {
              // RÃ¼ckwÃ¤rts-Ãœberlauf beendet!
              // Jetzt schalten wir den Zustand wieder frei, damit der Motor regulÃ¤r
              // und spielfrei von hinten (in positive Richtung) anfahren kann.
              Serial.println("[MOTOR] RÃ¼ckwÃ¤rts-Ãœberlauf beendet. Fahre jetzt spielfrei vorwÃ¤rts an.");
            }
        }
      return;
    }

  // Zustand B: Berechne den aktuellen Positionsfehler
  int32_t positionError = finalTarget - actualPos;

  if (positionError > 0)
    {
      // VorwÃ¤rtsfahrt (schwerer): Direkt und ohne Umwege anfahren
      driveMotorPWM(true, false);
    }
  else if (positionError < 0)
    {
      // RÃ¼ckwÃ¤rtsfahrt (leichter): Getriebespiel-Kompensation einleiten!
      // Berechne die temporÃ¤re Ãœberlauf-Position (Sollwert minus 32 Schritte)
      backlashTargetPos = finalTarget - (int32_t)MOTOR_NEG_RELPOS;
      if (backlashTargetPos < 0)
        backlashTargetPos = 0;

      // Wir wechseln aktiv in den BACKLASH-Zustand und bremsen die aktuelle Bewegung ab
      gpioData.currentMotorState = MotorState::BACKLASH;
      driveMotorPWM(false, true);
      Serial.printf("[MOTOR] Getriebespiel-Zustand aktiv! Fahre Ã¼berlaufend zurÃ¼ck auf: %d\n", backlashTargetPos);
    }
  else
    {
      // Zielpunkt exakt und spielfrei erreicht -> Sanft abbremsen und Haltestrom halten
      driveMotorPWM(true, true);
    }
}

void setFTMSOutputs()
{
  // Wir senden nur Daten, wenn ein Smartphone/Tablet aktiv verbunden ist
  if (!systemData.btClientConnected || pBikeDataCharacteristic == nullptr)
    {
      return;
    }

  static uint32_t lastTxTime = 0;
  if (millis() - lastTxTime >= 1000) // Exakt jede Sekunde senden
    {
      lastTxTime = millis();

      // FTMS Indoor Bike Data Format (Standard-Definition):
      // Byte 0-1: Flags (bestimmen, welche Daten gesendet werden)
      //   Bit 0: 0 = More Data present (wir senden alles in einem Paket)
      //   Bit 1: 1 = Average Speed present (nein, wir senden nur die aktuelle)
      //   Bit 2: 1 = Instantaneous Cadence present (Ja! Trittfrequenz ist dabei)
      //   Bit 3: 1 = Total Distance present (optional)
      //   Bit 4: 1 = Resistance Level present (optional)
      //   Bit 5: 1 = Instantaneous Power present (Ja! Leistung ist dabei)
      // Flags-Wert: Bit 2 (Cadence) und Bit 5 (Power) aktiv -> 0x0004 | 0x0020 = 0x0024
      // Da wir die Geschwindigkeit immer mitsenden (ist Standard, wenn Flags Bit 0 = 0):

      uint8_t txBuffer[8];

      // 1. Flags eintragen (Byte 0 & 1)
      uint16_t flags = 0x0024;
      txBuffer[0] = (uint8_t)(flags & 0xFF);
      txBuffer[1] = (uint8_t)((flags >> 8) & 0xFF);

      // 2. Geschwindigkeit eintragen (Byte 2 & 3)
      // FTMS-Standard verlangt km/h mit einer AuflÃ¶sung von 0.01 (km/h * 100)
      uint16_t speedVal = (uint16_t)(systemData.speedKmH * 100.0f);
      txBuffer[2] = (uint8_t)(speedVal & 0xFF);
      txBuffer[3] = (uint8_t)((speedVal >> 8) & 0xFF);

      // 3. Trittfrequenz eintragen (Byte 4 & 5)
      // FTMS-Standard verlangt Kurbelumdrehungen mit einer AuflÃ¶sung von 0.5 (RPM * 2)
      uint16_t cadenceVal = (uint16_t)(systemData.cadenceRPM * 2);
      txBuffer[4] = (uint8_t)(cadenceVal & 0xFF);
      txBuffer[5] = (uint8_t)((cadenceVal >> 8) & 0xFF);

      // 4. Leistung eintragen (Byte 6 & 7)
      // FTMS-Standard verlangt Watt als vorzeichenbehafteten 16-Bit Wert (Direktwert)
      int16_t powerVal = (int16_t)systemData.targetPowerWatt; // Oder Ist-Watt, sobald berechnet
      txBuffer[6] = (uint8_t)(powerVal & 0xFF);
      txBuffer[7] = (uint8_t)((powerVal >> 8) & 0xFF);

      // Daten an den Bluetooth-Stack Ã¼bergeben und an Kinomap abschicken!
      pBikeDataCharacteristic->setValue(txBuffer, 8);
      pBikeDataCharacteristic->notify(); // LÃ¶st das Senden auf Core 0 aus!
    }
}

void setup()
{
  ESP_init();
  Display_init();
  WLAN_init();
  Bluetooth_init();

  // Ergometer an Core 1 pinnen (Echtzeitsteuerung)
  xTaskCreatePinnedToCore(ErgometerLoop, "ErgometerTask", 10000, NULL, 1, &ErgometerTask, 1);

  // Gesamte kabellose Kommunikation an Core 0 pinnen (WLAN & Bluetooth)
  xTaskCreatePinnedToCore(wirelessConnectivityLoop, "WirelessTask", 4000, NULL, 1, &WirelessTask, 0);
}

void loop()
{
  vTaskDelay(pdMS_TO_TICKS(1000));
}

// Task 1: LÃ¤uft auf Core 1 (Reiner Echtzeit-Ergometer-Kram)
void ErgometerLoop(void *pvParameters)
{
  Serial.print("Ergometer-Task gestartet auf Core: ");
  Serial.println(xPortGetCoreID());
  for (;;)
    {
      main_prg();
      vTaskDelay(pdMS_TO_TICKS(10));
    }
}

// Task 2: LÃ¤uft im Hintergrund auf Core 0 (WLAN OTA & BLE Pulsgurt)
void wirelessConnectivityLoop(void *pvParameters)
{
  Serial.print("Wireless-Task gestartet auf Core: ");
  Serial.println(xPortGetCoreID());
  for (;;)
    {
      handleWlanOTA();                // Lauscht auf Core 0 nach OTA-Updates
      handleHeartRateConnection();    // Behandelt Pulsgurt-Verbindungen auf Core 0
      vTaskDelay(pdMS_TO_TICKS(100)); // 100ms CPU-Entlastung fÃ¼r Core 0
    }
}
