#include "config.h"
#include "modules.h"
#include "system_data.h"

void showDiagData()
{
  static uint32_t lastDiagTime = 0;
  if (millis() - lastDiagTime > DIAG_INTERVAL)
    {
      lastDiagTime = millis();
      Serial.printf("[DIAG - Core %d] Puls: %d bpm | Gang: %d | Power: %d W\n",
                    xPortGetCoreID(), systemData.heartRateBPM,
                    systemData.currentGear, systemData.currentPowerWatt);
    }
}

TaskHandle_t ErgometerTask;

// Der eigentliche Programmlauf fÃ¼r dein Ergometer auf Core 1
void ErgometerLoop(void *pvParameters)
{
  Serial.print("Ergometer-Task gestartet auf Core: ");
  Serial.println(xPortGetCoreID());

  for (;;)
    {
      main_prg();
      vTaskDelay(pdMS_TO_TICKS(10));
    }
}

void setup()
{
  ESP_init();
  WLAN_init();
  Bluetooth_init();

  xTaskCreatePinnedToCore(ErgometerLoop, "ErgometerTask", 10000, NULL, 1,
                          &ErgometerTask, 1);
}

void loop() { vTaskDelay(pdMS_TO_TICKS(1000)); }
